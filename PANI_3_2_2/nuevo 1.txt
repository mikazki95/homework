ISR (USART_RX_vect)
{
	comando[rx_pos]=UDR0;
	rx_pos++;

	if ((comando[0]=='M')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			inicio_med=1;
			flag_cancela=0;
			send_char('I');
			rx_pos=0;
			PORTB = 0b00000010;	//activa motor y válvula
		}
	}
	if ((comando[0]=='D')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			inicio_med=2;
			inicio_calc=0;
		//	send_char('F');
			rx_pos=0;
		}
	}
	if ((comando[0]=='V')&&(comando[1]=='\n'))		// cerrar válvula
	{
		if (rx_pos>=2)
		{
			PORTB &=~(0xFE);
			rx_pos=0;
			send_char('V');
			send_char('\n');
			flag_cancela=1;
		}
	}

	if ((comando[0]=='A')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			limite_presion=190;
			rx_pos=0;
			send_char('A');
			send_char('\n');
		}
	}
	if ((comando[0]=='N')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			limite_presion=160;
			rx_pos=0;
			send_char('N');
			send_char('\n');
		}
	}
	if ((comando[0]=='B')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			limite_presion=120;
			send_char('B');
			send_char('\n');
			rx_pos=0;
		}
	}
	if ((comando[0]=='I')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			send_string("I:PNI,");
			send_string(version_pani);
			send_char('\n');
			rx_pos=0;
		}
	}
	if ((comando[0]=='P')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			send_char('P');
			send_char('\n');
			rx_pos=0;
		}
	}
}
