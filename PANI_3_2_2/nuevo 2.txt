#define NPA700_R    0x51
#define NPA700_W    0x50

char version_pani[10]={"2.3.0"};	

char	comando[10]={0};

char	pres_str[10]={0};
char	m_pres_str[10]={0};
char	sist_str[10]={0};
char	valor_p[10]={0};
char	valor_m[10]={0};

int		pres_array[200]={0};//,pres_array1[230]={0};
int		delta_array[200]={0};//,m_array1[230]={0};
int		timer_array[200]={0};
int		limite_presion=190;

int cont_timer=0;

int8_t	inicio_med=0,rx_pos=0,inicio_calc=0;
int8_t	flag_cancela=0;

ISR (USART_RX_vect)
{
	comando[rx_pos]=UDR0;
	rx_pos++;

	if ((comando[0]=='M')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			inicio_med=1;
			flag_cancela=0;
			send_char('I');
			rx_pos=0;
			PORTB = 0b00000010;	//activa motor y válvula
		}
	}
	if ((comando[0]=='D')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			inicio_med=2;
			inicio_calc=0;
		//	send_char('F');
			rx_pos=0;
		}
	}
	if ((comando[0]=='V')&&(comando[1]=='\n'))		// cerrar válvula
	{
		if (rx_pos>=2)
		{
			PORTB &=~(0xFE);
			rx_pos=0;
			send_char('V');
			send_char('\n');
			flag_cancela=1;
		}
	}

	if ((comando[0]=='A')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			limite_presion=190;
			rx_pos=0;
			send_char('A');
			send_char('\n');
		}
	}
	if ((comando[0]=='N')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			limite_presion=160;
			rx_pos=0;
			send_char('N');
			send_char('\n');
		}
	}
	if ((comando[0]=='B')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			limite_presion=120;
			send_char('B');
			send_char('\n');
			rx_pos=0;
		}
	}
	if ((comando[0]=='I')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			send_string("I:PNI,");
			send_string(version_pani);
			send_char('\n');
			rx_pos=0;
		}
	}
	if ((comando[0]=='P')&&(comando[1]=='\n'))
	{
		if (rx_pos>=2)
		{
			send_char('P');
			send_char('\n');
			rx_pos=0;
		}
	}
}

ISR (TIMER0_OVF_vect)
{
	cont_timer++;
	TCNT0 = 184;
	
}

int main(void)
{
float delta_act=0;
float delta_max=0;
float suma_BPM=0;
float dif_pres=0;
int flag_max=0;
int p_dias=0;
int flag_sist=0;
float limi_dias=0, BPM=0;
float limi_sist=0;
int indice_max=0;
int max_delta=0;
int aux_dif=0;
int p_act_int=0;


	float	pres_npa=0;
	float	m_pres=0;
	float	pres_ant=0,max_pres=0,min_pres=0;
	float	pres_act=0, pres_min=0;
	float	p_sist=0,m_sist=0;

	int flag_start=0;
	int	i_array=0, indice=0;

	DDRB = 0b00000011;		//PB0  válvula   PB1 motor
	
	TCNT0 = 184;
	TCCR0A = 0x00;
	TCCR0B = 0x00;
	TIMSK0 = 0x01;

	cli();
		
	usart_init();

	UCSR0B =  (1<<RXEN0) | (1<<TXEN0)|(1<<RXCIE0);
	
	sei();
    
	i2c_inicio();	

	read_NPA700est(NPA700_R);

	_delay_ms(100);

	send_char('e');
    while (1) 
    {
		if (inicio_med==1)
 		{
	 		pres_npa=read_NPA700(NPA700_R);
			pres_act=pres_npa;

			if (pres_act>=limite_presion)
			{
				inicio_calc=1;
			}
			
//

			m_pres=(pres_act-pres_ant)/0.025;
			if (inicio_calc==1)
			{
				if (flag_start==0)
				{
					PORTB = 0b00000000;	//desctiva motor y válvula
					_delay_ms(500);
					flag_start=1;
					pres_min=pres_act;
					TCCR0B = 0X05;
					send_string("Inicio de calculo: ");
					send_char('\n');
					
					max_delta=0;
					
				} 
				else 
				{
					if (pres_act<pres_min)
					{
						pres_min=pres_act;
						
						if (flag_start==2)
						{

							timer_array[i_array]=cont_timer;
							delta_act=(float)delta_array[i_array];
							delta_max=(float)delta_array[indice_max]*0.3;
							
							if (delta_act<delta_max)
							{
								inicio_med=2;
								inicio_calc=0;
							}
							i_array++;
							flag_start=1;
							cont_timer=0;
											
						}
						
					}
					else 
					{
						
						aux_dif=(int)10*(pres_act-pres_min);
						if (delta_array[i_array]<aux_dif)
						{
							flag_start=2;
							indice=i_array;
							delta_array[i_array]=aux_dif;
							pres_array[i_array]=(int)pres_act;
							if (delta_array[i_array]>max_delta)
							{
								max_delta=delta_array[i_array];
								indice_max=i_array;
							}
						}
					}
				}
				
			}

			pres_ant=pres_act;
			
	
			//
			_delay_ms(25);  //valor ininial de puebas
 		}
		else if (inicio_med==2)
		{

			flag_sist=0;	
			flag_start=0;
			//send_string("numero de muestras: ");
			TCCR0B = 0x00;
			indice=i_array;
			/*
			for (int i=0;i<=i_array;i++)
			{
				if (delta_array[i]>max_delta)
				{
					max_delta=delta_array[i];
					indice_max=i;
				}
	

			}*/

			limi_sist=(float)delta_array[indice_max]*0.495;
			limi_dias=(float)delta_array[indice_max]*0.70;
			
			for (int j=0;j<=i_array;j++)
			{

// 				}
				if ((flag_sist==0)&&(delta_array[j]>limi_sist))
				{
					flag_sist=1;
					p_sist=pres_array[j-1];
				}
				if (j==i_array)
				{
					p_dias=pres_array[j];
				}
				else if ((j>indice_max)&&(delta_array[j]<limi_dias))//&&(flag_diast==0))
				{
					
					p_dias=pres_array[j-1];
					j=i_array+1;
				}
			}

			for (int i=1;i<=i_array;i++)
			{
				suma_BPM+=0.3*(float)timer_array[i];
				p_sist=(float)timer_array[i];
				//dtostrf(p_sist,4,2,valor_m);
				//send_string(valor_m);
				//send_char('\n');
							//
			}
			
			send_string("Sistólica: ");
			dtostrf(p_sist,4,2,sist_str);
			send_string(sist_str);
			send_char('\n');

			send_string("Diastólica: ");
			dtostrf(p_dias,4,2,valor_m);
			send_string(valor_m);
			send_char('\n');

			//p_sist=(float)i_array;
			//send_string("i_array: ");
			//dtostrf(p_sist,4,2,sist_str);
			//send_string(sist_str);
			//send_char('\n');
			
			BPM=suma_BPM/(i_array);
			send_string("BPM: ");
			dtostrf(BPM,2,2,valor_m);
			send_string(valor_m);
			send_char('\n');

			for (int j=0;j<=indice;j++)
			{
				pres_array[j]=0;
				delta_array[j]=0;
			}
			suma_BPM=0;
			BPM=0;
			i_array=0;
			inicio_med=0;
			min_pres=0;
			max_pres=0;
			dif_pres=0;
			flag_max=0;
			PORTB = 0b00000001;
			_delay_ms(5000);
			PORTB = 0x00;
		}
		
	}		
}
